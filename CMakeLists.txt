cmake_minimum_required(VERSION 3.8)
project(pci_general)
# ROS 2 (ament) build of the GBPlanner port; sources are the ROS 1 ones with the ROS API ported.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-Wall -Wno-sign-compare -Wno-unused-variable -Wno-unused-but-set-variable -Wno-format -Wno-unused-parameter -Wno-reorder -Werror=return-type)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ros1_compat REQUIRED)
find_package(planner_msgs REQUIRED)
find_package(planner_control_interface REQUIRED)
find_package(mav_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(Eigen3 REQUIRED)
set(EXTRA_LIBS glog gflags)
# mav_comm (ros2 branch) installs its helper headers to <prefix>/include/mav_msgs, while rosidl on
# Jazzy exports <prefix>/include/mav_msgs as the include directory; add <prefix>/include as well.
get_filename_component(MAV_MSGS_PREFIX "${mav_msgs_DIR}/../../.." ABSOLUTE)
include_directories(${MAV_MSGS_PREFIX}/include)

add_library(${PROJECT_NAME} SHARED src/pci_general.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${PCL_LIBRARIES} ${EXTRA_LIBS})
ament_target_dependencies(${PROJECT_NAME} rclcpp ros1_compat planner_msgs planner_control_interface mav_msgs trajectory_msgs geometry_msgs nav_msgs visualization_msgs std_srvs tf2)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)

add_executable(pci_general_ros_node src/pci_general_ros_node.cpp)
target_include_directories(pci_general_ros_node PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include ${PCL_INCLUDE_DIRS})
target_link_libraries(pci_general_ros_node ${PROJECT_NAME} ${PCL_LIBRARIES} ${EXTRA_LIBS})
ament_target_dependencies(pci_general_ros_node rclcpp ros1_compat planner_msgs planner_control_interface mav_msgs trajectory_msgs geometry_msgs nav_msgs visualization_msgs std_srvs tf2)
install(TARGETS pci_general_ros_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include OPTIONAL)
ament_export_include_directories(include)
ament_export_dependencies(rclcpp ros1_compat planner_msgs planner_control_interface mav_msgs trajectory_msgs geometry_msgs nav_msgs visualization_msgs std_srvs tf2)
ament_package()
